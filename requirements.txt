# Eidolon Society Lab — Python dependencies
#
# Core simulation and the live Society Lab UI need no pip packages:
#   python3 main.py
#   python3 stress_test.py
#   python3 record_demo.py
#   python3 live_server.py
# Those entry points use the Python 3.9+ standard library only.
#
# The analytics page loads Chart.js 4.4.1 from cdnjs in the browser.
# That is not a Python dependency.
#
# LLM-backed agents (main_llm.py, or live_server.py --llm) need Groq.
# Install into the same interpreter you will run:
#   python3 -m pip install -r requirements.txt
# Bare `pip` can target Anaconda while `python3` is a different binary.

groq>=1.5.0
